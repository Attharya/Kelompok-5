	
		ALGORITHM "SIMPLE ATM SIMULATOR"


START

    saldo ← 1000000

    REPEAT

		DISPLAY "========================="
        DISPLAY "===== ATM SEDERHANA ====="
		DISPLAY "========================="
        DISPLAY "1. Cek Saldo"
        DISPLAY "2. Setor Uang"
        DISPLAY "3. Tarik Uang"
        DISPLAY "4. Keluar"
		DISPLAY "========================="
        DISPLAY "Pilih menu: "

        INPUT pilihan
        SWITCH pilihan

            IF pilihan = "1" THEN
            	DISPLAY "Saldo Anda: Rp ", saldo

            ELSE IF pilihan = "2" THEN
            	DISPLAY "Masukkan jumlah uang yang ingin disetor:"
            	INPUT jumlah

				IF jumlah > 0 THEN
                	saldo ← saldo + jumlah
                	DISPLAY "Setor uang berhasil."
                	DISPLAY "Saldo Anda: Rp ", saldo
				ELSE 
					DISPLAY "Jumlah setoran harus lebih dari 0."
				END IF


           	ELSE IF pilihan = "3" THEN
				DISPLAY "Masukkan jumlah uang yang ingin ditarik : "
				INPUT jumlah

				IF jumlah > 0 AND jumlah <= saldo THEN
                	saldo ← saldo - jumlah
                	DISPLAY "Penarikan uang berhasil."
                	DISPLAY "Sisa saldo Anda: Rp ", saldo
				ELSE IF jumlah <=0 THEN
					DISPLAY "Jumlah penarikan harus lebih dari 0."
				ELSE
                	DISPLAY "Saldo tidak mencukupi."
            	END IF

			ELSE IF pilihan ="4' THEN
				DISPLAY "Terima kasih telah menggunakan ATM Sederhana."
				DISPLAY "Saldo akhir anda : Rp ", saldo
			END IF

			//Konfirmasi Lanjut Transaksi
			IF pilihan !== "4" THEN
				DISPLAY "Apakah Anda ingin melakukan transaksi lain? (Y/N)"
				INPUT transaksi
				transaksi <-- TO_UPPERCASE(transaksi)

				IF transaksi = "N" THEN
					DISPLAY "Transaksi selesai."
                	DISPLAY "Saldo akhir Anda: Rp ", saldo
				END IF
			END IF

    UNTIL pilihan = "4" OR transaksi = "N"
END
